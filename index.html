import math
import warnings

import numpy as np
import pandas as pd
import patsy
import scipy.stats as stats
import statsmodels.api as sm
import statsmodels.formula.api as smf
from statsmodels.stats.multitest import multipletests

warnings.filterwarnings("ignore")

OUTCOMES = ["SHAPS", "HAMD", "HAMA", "BDI", "BAI"]
FOLLOWUPS = ["3M", "6M"]
VISITS = ["0W", "3M", "6M"]
N_ITEMS = {"SHAPS": 14, "HAMD": 17, "HAMA": 14, "BDI": 21, "BAI": 21}
REDUCED = ["HAMD_m7", "HAMD_m7_14", "BDI_m4", "BDI_anh"]
CONFOUNDERS = ["Age", "sex_male", "Education_years", "log_total_illness", "log_current_episode",
               "log_untreated", "SHAPS_0W", "HAMD_0W", "HAMA_0W", "BDI_0W", "BAI_0W", "FE_0W"]


def items(scale):
    return [f"{scale}_{i:02d}" for i in range(1, N_ITEMS[scale] + 1)]


def score(scales):
    d = scales.copy()
    for s in OUTCOMES:
        d[s] = d[items(s)].sum(axis=1)
    d["HAMD_m7"] = d["HAMD"] - d["HAMD_07"]
    d["HAMD_m7_14"] = d["HAMD"] - d["HAMD_07"] - d["HAMD_14"]
    d["BDI_anh"] = d[["BDI_04", "BDI_12", "BDI_15", "BDI_21"]].sum(axis=1)
    d["BDI_m4"] = d["BDI"] - d["BDI_anh"]
    d["SHAPS_dich"] = (d[items("SHAPS")] >= 3).sum(axis=1)
    return d


def build_wide(participants, scored):
    w = participants.copy()
    for v in VISITS:
        x = scored[scored.visit == v].set_index("ID")
        for c in OUTCOMES + REDUCED + ["SHAPS_dich"]:
            w[f"{c}_{v}"] = w.ID.map(x[c])
    w["observed_3M"] = w.SHAPS_3M.notna()
    w["observed_6M"] = w.SHAPS_6M.notna()
    w["analysis_sample"] = w.observed_3M | w.observed_6M
    w["log_total_illness"] = np.log1p(w.Total_illness_duration_months)
    w["log_current_episode"] = np.log1p(w.Current_episode_duration_months)
    w["log_untreated"] = np.log1p(w.Untreated_duration_months)
    return w


def mean_sd(x):
    x = pd.to_numeric(x, errors="coerce").dropna()
    return x.mean(), x.std(ddof=1), len(x)


def median_iqr(x):
    x = pd.to_numeric(x, errors="coerce").dropna()
    return x.median(), x.quantile(0.25), x.quantile(0.75)


def smd(a, b):
    a = pd.to_numeric(a, errors="coerce").dropna()
    b = pd.to_numeric(b, errors="coerce").dropna()
    pooled = math.sqrt((a.var(ddof=1) + b.var(ddof=1)) / 2)
    return abs(a.mean() - b.mean()) / pooled if pooled else np.nan


def weighted_smd(x, g, w):
    m1 = np.average(x[g == 1], weights=w[g == 1])
    m0 = np.average(x[g == 0], weights=w[g == 0])
    v1 = np.average((x[g == 1] - m1) ** 2, weights=w[g == 1])
    v0 = np.average((x[g == 0] - m0) ** 2, weights=w[g == 0])
    return abs(m1 - m0) / math.sqrt((v1 + v0) / 2)


def table1(w):
    rows = []
    t, c = w[w.group_treat == 1], w[w.group_treat == 0]
    for v in ["Age", "Education_years", "Onset_age", "FE_0W", "SHAPS_0W", "HAMD_0W", "HAMA_0W", "BDI_0W", "BAI_0W"]:
        rows.append({"variable": v, "buspirone": mean_sd(t[v])[:2], "control": mean_sd(c[v])[:2],
                     "smd": smd(t[v], c[v]), "p": stats.ttest_ind(t[v], c[v], nan_policy="omit").pvalue})
    for v in ["Current_episode_duration_months", "Total_illness_duration_months", "Untreated_duration_months"]:
        rows.append({"variable": v, "buspirone": median_iqr(t[v]), "control": median_iqr(c[v]),
                     "smd": smd(t[v], c[v]), "p": stats.mannwhitneyu(t[v].dropna(), c[v].dropna(), alternative="two-sided").pvalue})
    for v in ["sex_male", "Benzodiazepine_present_0W", "Zdrug_or_hypnotic_present_0W"]:
        tab = pd.crosstab(w[v], w.group_treat)
        p = stats.fisher_exact(tab.values)[1] if tab.shape == (2, 2) and tab.values.min() < 5 else stats.chi2_contingency(tab)[1]
        rows.append({"variable": v, "buspirone": int(t[v].sum()), "control": int(c[v].sum()), "smd": smd(t[v], c[v]), "p": p})
    for v in ["Center", "SSRI_type_0W"]:
        tab = pd.crosstab(w[v], w.group_treat)
        rows.append({"variable": v, "buspirone": tab[1].to_dict(), "control": tab[0].to_dict(), "smd": np.nan,
                     "p": stats.chi2_contingency(tab)[1]})
    return pd.DataFrame(rows)


def baseline_difference(w, outcome):
    a = w.loc[w.group_treat == 1, f"{outcome}_0W"].dropna()
    b = w.loc[w.group_treat == 0, f"{outcome}_0W"].dropna()
    diff = a.mean() - b.mean()
    df = len(a) + len(b) - 2
    sp = math.sqrt(((len(a) - 1) * a.var(ddof=1) + (len(b) - 1) * b.var(ddof=1)) / df)
    se = sp * math.sqrt(1 / len(a) + 1 / len(b))
    tcrit = stats.t.ppf(0.975, df)
    return {"outcome": outcome, "difference": diff, "ci_low": diff - tcrit * se, "ci_high": diff + tcrit * se,
            "p": stats.ttest_ind(a, b).pvalue}


def make_long(w, outcome, ids=None, extra=()):
    src = w[w.analysis_sample]
    if ids is not None:
        src = src[src.ID.isin(set(ids))]
    rows = []
    for tp in FOLLOWUPS:
        cols = ["ID", "group_treat", "Center", "FE_0W", f"FE_{tp}", f"{outcome}_0W", f"{outcome}_{tp}"] + list(extra)
        cols = list(dict.fromkeys(cols))
        x = src[cols].rename(columns={f"{outcome}_{tp}": "y", f"{outcome}_0W": "baseline", f"FE_{tp}": "FE_time"}).copy()
        x["time"] = tp
        rows.append(x)
    long = pd.concat(rows, ignore_index=True).dropna(subset=["y", "baseline", "FE_0W"])
    long["time"] = pd.Categorical(long["time"], categories=FOLLOWUPS)
    return long


def fit_lmm(long, formula):
    model = smf.mixedlm(formula, data=long, groups=long["ID"])
    for method in ["lbfgs", "bfgs", "cg", "powell", "nm"]:
        try:
            res = model.fit(reml=False, method=method, maxiter=1000, disp=False)
            if res.converged and np.isfinite(res.llf) and float(res.cov_re.iloc[0, 0]) > 0:
                return res
        except Exception:
            continue
    raise RuntimeError(formula)


def contrasts(res, long):
    info = res.model.data.design_info
    params = res.fe_params
    names = list(params.index)
    cov = res.cov_params().loc[names, names]
    out = {}
    for tp in FOLLOWUPS:
        base = long[long["time"] == tp].copy()
        vec = {}
        for g in (0, 1):
            new = base.copy()
            new["group_treat"] = g
            new["time"] = pd.Categorical([tp] * len(new), categories=FOLLOWUPS)
            vec[g] = patsy.build_design_matrices([info], new, return_type="dataframe")[0][names].mean(axis=0)
        c = vec[1] - vec[0]
        est = float(c @ params)
        se = float(np.sqrt(c @ cov @ c))
        out[tp] = {"estimate": est, "se": se, "ci_low": est - 1.96 * se, "ci_high": est + 1.96 * se,
                   "p": 2 * stats.norm.sf(abs(est / se))}
    return out


def lmm(w, outcome, fe="FE_0W", ids=None, covariates=()):
    long = make_long(w, outcome, ids, extra=covariates)
    fe_term = "FE_time" if fe == "time" else "FE_0W"
    formula = f"y ~ group_treat * C(time) + baseline + C(Center) + {fe_term}"
    if covariates:
        formula += " + " + " + ".join(c for c in covariates if c != f"{outcome}_0W")
    if fe == "time":
        long = long.dropna(subset=["FE_time"])
    res = fit_lmm(long, formula)
    out = contrasts(res, long)
    out["interaction_p"] = float(res.pvalues["group_treat:C(time)[T.6M]"])
    out["n_subjects"] = long.ID.nunique()
    return out


def main_models(w):
    res = {o: lmm(w, o) for o in OUTCOMES}
    rows = [{"outcome": o, "time": tp, **res[o][tp]} for o in OUTCOMES for tp in FOLLOWUPS]
    q = multipletests([r["p"] for r in rows], method="fdr_bh")[1]
    for r, qq in zip(rows, q):
        r["q"] = qq
    return res, pd.DataFrame(rows)


def cohen_d_change(w, outcome, estimate, tp):
    d = w.dropna(subset=[f"{outcome}_{tp}", f"{outcome}_0W"])
    ch = d[f"{outcome}_{tp}"] - d[f"{outcome}_0W"]
    a, b = ch[d.group_treat == 1], ch[d.group_treat == 0]
    pooled = math.sqrt(((len(a) - 1) * a.var(ddof=1) + (len(b) - 1) * b.var(ddof=1)) / (len(a) + len(b) - 2))
    return estimate / pooled


def ancova(w, outcome, tp, ids=None):
    d = w if ids is None else w[w.ID.isin(set(ids))]
    d = d.dropna(subset=[f"{outcome}_{tp}", f"{outcome}_0W", "FE_0W"])
    r = smf.ols(f"{outcome}_{tp} ~ group_treat + {outcome}_0W + C(Center) + FE_0W", d).fit(cov_type="HC3")
    e, se = float(r.params.group_treat), float(r.bse.group_treat)
    return {"estimate": e, "ci_low": e - 1.96 * se, "ci_high": e + 1.96 * se, "p": float(r.pvalues.group_treat), "n": int(r.nobs)}


def sensitivity(w):
    out = {"time_varying_FE": lmm(w, "SHAPS", fe="time")}
    out["complete_cases"] = lmm(w, "SHAPS", ids=w.loc[w.observed_3M & w.observed_6M, "ID"])
    out["no_benzo_zdrug"] = lmm(w, "SHAPS", ids=w.loc[(w.Benzodiazepine_present_0W.fillna(0) == 0)
                                                      & (w.Zdrug_or_hypnotic_present_0W.fillna(0) == 0), "ID"])
    out["FE_20_40"] = lmm(w, "SHAPS", ids=w.loc[w.FE_0W.between(20, 40), "ID"])
    for c in ["HX", "ST", "JY", "NC"]:
        out[f"without_{c}"] = lmm(w, "SHAPS", ids=w.loc[w.Center != c, "ID"])
    severe = w.loc[w.HAMD_0W > 17, "ID"]
    out["ancova_full"] = {tp: ancova(w, "SHAPS", tp) for tp in FOLLOWUPS}
    out["ancova_HAMD_gt17"] = {tp: ancova(w, "SHAPS", tp, severe) for tp in FOLLOWUPS}
    return out


def missingness(w):
    d = w.copy()
    d["missing_6M"] = (~d.observed_6M).astype(int)
    r = smf.logit("missing_6M ~ group_treat + Age + sex_male + SHAPS_0W + HAMD_0W + HAMA_0W + FE_0W + C(Center)", d).fit(disp=False)
    ci = r.conf_int()
    return pd.DataFrame({"OR": np.exp(r.params), "ci_low": np.exp(ci[0]), "ci_high": np.exp(ci[1]), "p": r.pvalues})


def rubin(est, var):
    q, u = np.asarray(est), np.asarray(var)
    m = len(q)
    qbar, ubar, b = q.mean(), u.mean(), q.var(ddof=1)
    total = ubar + (1 + 1 / m) * b
    r = (1 + 1 / m) * b / ubar
    df = (m - 1) * (1 + 1 / r) ** 2
    se = math.sqrt(total)
    t = stats.t.ppf(0.975, df)
    return {"estimate": qbar, "ci_low": qbar - t * se, "ci_high": qbar + t * se, "p": 2 * stats.t.sf(abs(qbar / se), df)}


def tipping_point(w, m=100, seed=20260716, deltas=range(0, 13)):
    rng = np.random.default_rng(seed)
    formula = "SHAPS_6M ~ group_treat + SHAPS_0W + C(Center) + FE_0W"
    obs = w.dropna(subset=["SHAPS_6M", "SHAPS_0W", "FE_0W"])
    mis = w[w.SHAPS_6M.isna()].dropna(subset=["SHAPS_0W", "FE_0W"])
    fit = smf.ols(formula, obs).fit()
    xm = np.asarray(patsy.build_design_matrices([fit.model.data.design_info], mis, return_type="dataframe")[0])
    beta, xtx_inv, df = np.asarray(fit.params), np.asarray(fit.normalized_cov_params), int(fit.df_resid)
    s2 = float(fit.ssr / df)
    imputations = []
    for _ in range(m):
        sigma2 = df * s2 / rng.chisquare(df)
        b = rng.multivariate_normal(beta, sigma2 * xtx_inv)
        imputations.append(xm @ b + rng.normal(0, math.sqrt(sigma2), len(mis)))
    treated = mis.group_treat.to_numpy() == 1
    out = []
    for delta in deltas:
        est, var = [], []
        for y in imputations:
            d = w.copy()
            vals = y.copy()
            vals[treated] += delta
            d.loc[mis.index, "SHAPS_6M"] = vals
            r = smf.ols(formula, d).fit(cov_type="HC3")
            est.append(float(r.params.group_treat))
            var.append(float(r.cov_params().loc["group_treat", "group_treat"]))
        out.append({"delta": delta, **rubin(est, var)})
    return pd.DataFrame(out)


def gee_contrasts(res):
    b, v = res.params, res.cov_params()
    out = {}
    for tp in FOLLOWUPS:
        c = pd.Series(0.0, index=b.index)
        c["group_treat"] = 1
        if tp == "6M":
            c["group_treat:C(time)[T.6M]"] = 1
        est = float(c @ b)
        se = float(np.sqrt(c @ v @ c))
        out[tp] = {"estimate": est, "ci_low": est - 1.96 * se, "ci_high": est + 1.96 * se, "p": 2 * stats.norm.sf(abs(est / se))}
    return out


def confounding(w):
    out = {"multivariable": lmm(w, "SHAPS", covariates=[c for c in CONFOUNDERS if c not in ("SHAPS_0W", "FE_0W")])}
    d = w.dropna(subset=CONFOUNDERS).copy()
    ps = smf.logit("group_treat ~ " + " + ".join(CONFOUNDERS) + " + C(Center)", d).fit(disp=False).predict(d)
    pt = d.group_treat.mean()
    d["iptw"] = np.where(d.group_treat == 1, pt / ps, (1 - pt) / (1 - ps))
    for c in d.Center.unique():
        d[f"center_{c}"] = (d.Center == c).astype(int)
    balance = []
    for v in CONFOUNDERS + [f"center_{c}" for c in sorted(d.Center.unique())]:
        balance.append({"variable": v, "smd_before": weighted_smd(d[v].to_numpy(), d.group_treat.to_numpy(), np.ones(len(d))),
                        "smd_after": weighted_smd(d[v].to_numpy(), d.group_treat.to_numpy(), d.iptw.to_numpy())})
    out["balance"] = pd.DataFrame(balance)
    long = make_long(d.assign(analysis_sample=d.analysis_sample), "SHAPS", extra=CONFOUNDERS + ["iptw"])
    long = long.rename(columns={"baseline": "SHAPS_base"})
    ind = sm.cov_struct.Independence()
    r1 = smf.gee("y ~ group_treat * C(time)", "ID", long, weights=long.iptw, cov_struct=ind).fit()
    covs = [c for c in CONFOUNDERS if c != "SHAPS_0W"] + ["SHAPS_base"]
    r2 = smf.gee("y ~ group_treat * C(time) + " + " + ".join(covs) + " + C(Center)", "ID", long, weights=long.iptw, cov_struct=ind).fit()
    out["iptw"] = gee_contrasts(r1)
    out["doubly_robust"] = gee_contrasts(r2)
    return out


def e_value(estimate, ci_limit, sd):
    def ev(d):
        rr = math.exp(0.91 * abs(d))
        return rr + math.sqrt(rr * (rr - 1))
    return {"point": ev(estimate / sd), "ci": ev(ci_limit / sd) if ci_limit < 0 else 1.0}


def remission(w):
    rows = []
    defs = [("SHAPS_dich", lambda x: x <= 2, True), ("HAMD", lambda x: x <= 7, False), ("HAMA", lambda x: x <= 7, False)]
    for var, rule, adjust in defs:
        for tp in FOLLOWUPS:
            d = w.dropna(subset=[f"{var}_{tp}"]).copy()
            d["rem"] = rule(d[f"{var}_{tp}"]).astype(int)
            a, b = d[d.group_treat == 1].rem, d[d.group_treat == 0].rem
            row = {"outcome": var, "time": tp, "buspirone": (int(a.sum()), len(a)), "control": (int(b.sum()), len(b)),
                   "p_unadjusted": stats.fisher_exact([[a.sum(), len(a) - a.sum()], [b.sum(), len(b) - b.sum()]])[1]}
            if adjust:
                r = smf.logit("rem ~ group_treat + SHAPS_0W + C(Center) + FE_0W", d).fit(disp=False)
                ci = np.exp(r.conf_int().loc["group_treat"])
                row.update({"OR": math.exp(r.params.group_treat), "ci_low": ci[0], "ci_high": ci[1], "p_adjusted": r.pvalues.group_treat})
            rows.append(row)
    return pd.DataFrame(rows)


def anxiety_baseline(w):
    a = (w.loc[w.group_treat == 1, "HAMA_0W"] >= 14)
    b = (w.loc[w.group_treat == 0, "HAMA_0W"] >= 14)
    return {"buspirone": (int(a.sum()), len(a)), "control": (int(b.sum()), len(b)),
            "p": stats.fisher_exact([[a.sum(), len(a) - a.sum()], [b.sum(), len(b) - b.sum()]])[1]}


def standardized_differences(w, outcomes):
    out = {}
    for o in outcomes:
        r = lmm(w, o)
        out[o] = {tp: r[tp]["estimate"] / w[f"{o}_{tp}"].std(ddof=1) for tp in FOLLOWUPS}
    return out


def specificity(w, n_boot=2000, seed=20261026):
    outcomes = ["SHAPS", "HAMD", "HAMD_m7", "HAMD_m7_14", "BDI", "BDI_m4", "BDI_anh", "HAMA", "BAI"]
    models = {o: lmm(w, o) for o in outcomes}
    sd = {o: {tp: models[o][tp]["estimate"] / w[f"{o}_{tp}"].std(ddof=1) for tp in FOLLOWUPS} for o in outcomes}
    compare = ["HAMD", "HAMD_m7", "HAMD_m7_14", "BDI", "BDI_m4", "HAMA", "BAI"]
    groups = {i: g for i, g in w.groupby("ID")}
    t_ids = w.loc[w.group_treat == 1, "ID"].to_numpy()
    c_ids = w.loc[w.group_treat == 0, "ID"].to_numpy()
    boot = []
    for k in range(n_boot):
        rng = np.random.default_rng(seed + k)
        pick = np.concatenate([rng.choice(t_ids, len(t_ids)), rng.choice(c_ids, len(c_ids))])
        bw = pd.concat([groups[i].assign(ID=f"b{j}") for j, i in enumerate(pick)], ignore_index=True)
        boot.append(standardized_differences(bw, ["SHAPS"] + compare))
    rows = []
    for tp in FOLLOWUPS:
        s = np.array([b["SHAPS"][tp] for b in boot])
        for o in compare:
            diff = s - np.array([b[o][tp] for b in boot])
            rows.append({"time": tp, "comparison": f"SHAPS vs {o}", "sd_SHAPS": sd["SHAPS"][tp], "sd_other": sd[o][tp],
                         "difference": sd["SHAPS"][tp] - sd[o][tp], "ci_low": np.percentile(diff, 2.5),
                         "ci_high": np.percentile(diff, 97.5), "p": 2 * min((diff >= 0).mean(), (diff <= 0).mean())})
    return models, sd, pd.DataFrame(rows)


def cronbach_alpha(x):
    x = np.asarray(x, float)
    k = x.shape[1]
    return k / (k - 1) * (1 - x.var(axis=0, ddof=1).sum() / x.sum(axis=1).var(ddof=1))


def reliability(scored):
    return pd.DataFrame([{"scale": s, "visit": v, "alpha": cronbach_alpha(scored.loc[scored.visit == v, items(s)])}
                         for s in OUTCOMES for v in VISITS])


def tess(tess_data, events):
    rows = []
    for tp in FOLLOWUPS:
        d = tess_data[tess_data.visit == tp]
        a, b = d[d.group_treat == 1], d[d.group_treat == 0]
        rows.append({"time": tp, "measure": "TESS_total", "p": stats.mannwhitneyu(a.TESS_total, b.TESS_total, alternative="two-sided").pvalue})
        for e in events:
            x1, n1, x0, n0 = int(a[e].sum()), len(a), int(b[e].sum()), len(b)
            rr = (x1 / n1) / (x0 / n0) if x0 else np.nan
            se = math.sqrt(1 / x1 - 1 / n1 + 1 / x0 - 1 / n0) if x1 and x0 else np.nan
            rows.append({"time": tp, "measure": e, "buspirone": (x1, n1), "control": (x0, n0), "RR": rr,
                         "ci_low": rr * math.exp(-1.96 * se), "ci_high": rr * math.exp(1.96 * se),
                         "p": stats.fisher_exact([[x1, n1 - x1], [x0, n0 - x0]])[1]})
    return pd.DataFrame(rows)


def main():
    participants = pd.read_csv("data/participants.csv")
    scales = pd.read_csv("data/scales.csv")
    tess_data = pd.read_csv("data/tess.csv").merge(participants[["ID", "group_treat"]], on="ID")
    scored = score(scales)
    w = build_wide(participants, scored)

    results = {"table1": table1(w),
               "baseline_differences": pd.DataFrame([baseline_difference(w, o) for o in OUTCOMES]),
               "anxiety_baseline": anxiety_baseline(w),
               "reliability": reliability(scored)}
    models, table2 = main_models(w)
    table2["cohen_d"] = [cohen_d_change(w, r.outcome, r.estimate, r.time) for r in table2.itertuples()]
    results.update({"table2": table2, "interaction_p": {o: models[o]["interaction_p"] for o in OUTCOMES},
                    "sensitivity": sensitivity(w), "missingness": missingness(w), "tipping_point": tipping_point(w)})
    conf = confounding(w)
    sd6 = w.SHAPS_6M.std(ddof=1)
    results["confounding"] = conf
    results["e_value"] = e_value(models["SHAPS"]["6M"]["estimate"], models["SHAPS"]["6M"]["ci_high"], sd6)
    results["remission"] = remission(w)
    spec_models, spec_sd, spec_boot = specificity(w)
    results.update({"specificity_models": spec_models, "standardized_differences": spec_sd, "specificity_bootstrap": spec_boot})
    results["tess"] = tess(tess_data, [c for c in tess_data.columns if c not in ("ID", "visit", "group_treat", "TESS_total")])
    return results
